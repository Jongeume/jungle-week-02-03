 # 제자리 회전
def rotate_matrix(matrix):
    n = len(matrix)
## 1. 전치 도출
[1,2,3]        [1,4,7]
[4,5,6]   →    [2,5,8]
[7,8,9]        [3,6,9]

for i in range(n):
    for j in range(i+1, n):
        matrix[i][j], matrix[j][i] = matrix[j][i], matrix[i][j]

## 2. 도출된 값 reverse
[1,4,7]       [7,4,1]
[2,5,8]   →   [8,5,2]
[3,6,9]       [9,6,3]
for row in matrix:
    row.reverse()
return matrix
